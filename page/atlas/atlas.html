ReportUsageProvider.java 
@Component
public class ReportUsageProvider implements EruptUsageProvider {

    @Resource
    private ReportRepository reportRepository;

    @Override
    public List<EruptUsage> usages() {
        return reportRepository.findAll().stream().map(r ->
            new EruptUsage(
                r.getErupt(),                      // {{'atlas.usage.c1'|i18n}}
                I18nTranslate.$translate("Report"), // {{'atlas.usage.c2'|i18n}}
                r.getTitle(),                      // {{'atlas.usage.c3'|i18n}}
                EruptUsage.tableRoute(Report.class) // {{'atlas.usage.c4'|i18n}}
            )).toList();
    }
}
